cmake_minimum_required(VERSION 3.25)
set(GAMES_BASE_VERSION "0.5.0")
set(GAMES_RELEASE_VERSION "$ENV{GAMES_RELEASE_VERSION}" CACHE STRING "Release version from the tested publication workflow")
if(GAMES_RELEASE_VERSION STREQUAL "")
  set(GAMES_RELEASE_VERSION "${GAMES_BASE_VERSION}")
endif()
if(NOT GAMES_RELEASE_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "GAMES_RELEASE_VERSION must be a numeric major.minor.patch")
endif()
project(Games VERSION ${GAMES_RELEASE_VERSION} LANGUAGES C CXX)
# Windows finds a program's DLLs beside it: build every program and DLL, GUI.Forms'
# Application and Threading included, into one folder.
if(WIN32)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endif()
set(GAMES_RUNTIME_ASSET_DIR "" CACHE PATH "Complete verified portable runtime assets")
option(GAMES_BUILD_APPLICATION "Build the complete graphical game collection and UI tests" ON)
enable_testing()
add_executable(dogfood_tests tests/dogfood_tests.cpp src/dogfood.cpp)
target_include_directories(dogfood_tests PRIVATE src)
target_compile_features(dogfood_tests PRIVATE cxx_std_20)
add_test(NAME dogfood COMMAND dogfood_tests)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
endif()
include(FetchContent)
FetchContent_Declare(quickjs
  GIT_REPOSITORY https://github.com/quickjs-ng/quickjs.git
  GIT_TAG 8eb27e39848b73885e092777f0a67a4a2d8e5b99)
set(QJS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(QJS_BUILD_LIBC OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(quickjs)
set_property(DIRECTORY ${quickjs_SOURCE_DIR} PROPERTY EXCLUDE_FROM_ALL TRUE)
if(GAMES_BUILD_APPLICATION)
  set(GAMES_TOOLKIT_SOURCE_DIR "" CACHE PATH "Reviewed GUI.Forms source for development application builds")
  if(NOT EXISTS "${GAMES_TOOLKIT_SOURCE_DIR}/include/gui_forms/text_mask.hpp")
    message(FATAL_ERROR "Set GAMES_TOOLKIT_SOURCE_DIR to the reviewed GUI.Forms source. Text masks and bar audio are not yet exported by the installed SDK; see docs/PORTABLE_TEXT_REQUIREMENTS.md.")
  endif()
  set(GUI_FORMS_BUILD_GALLERY OFF CACHE BOOL "" FORCE)
  set(GUI_FORMS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GUI_FORMS_BUILD_AUDIO ON CACHE BOOL "" FORCE)
  set(GUI_FORMS_BUILD_AUDIO_LOOP_TRANSPORT ON CACHE BOOL "" FORCE)
  # GUI.Forms' own Vorbis decoder; forced so an older build folder cannot keep stb.
  set(GUI_FORMS_AUDIO_VORBIS_BACKEND stx CACHE STRING "" FORCE)
  set(GUI_FORMS_BUILD_TEXT_MASKS ON CACHE BOOL "" FORCE)
  set(GUI_FORMS_ENABLE_HARFBUZZ_TEXT ON CACHE BOOL "" FORCE)
  if(WIN32)
    set(GUI_FORMS_ENABLE_SKIA OFF CACHE BOOL "" FORCE)
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(GUI_FORMS_ENABLE_LINUX_HOST ON CACHE BOOL "" FORCE)
  endif()
  add_subdirectory("${GAMES_TOOLKIT_SOURCE_DIR}" toolkit EXCLUDE_FROM_ALL)
  set(GUIForms_FONT_DIR "${GAMES_TOOLKIT_SOURCE_DIR}/assets/fonts")
endif()
add_library(game_paths src/runtime_paths.cpp)
target_include_directories(game_paths PUBLIC src)
add_library(game_rules shared/cards/game.cpp src/storage.cpp src/suite_settings.cpp src/scores.cpp games/sudoku/src/sudoku.cpp shared/puzzles/puzzles.cpp)
target_include_directories(game_rules PUBLIC src shared/cards shared/puzzles games/sudoku/src)
target_link_libraries(game_rules PUBLIC game_paths)
add_library(game_raster shared/puzzles/puzzle_render.cpp shared/puzzles/puzzle_image.cpp)
target_link_libraries(game_raster PUBLIC game_rules)
add_executable(raster_tests tests/raster_tests.cpp)
target_link_libraries(raster_tests PRIVATE game_raster)
add_test(NAME raster COMMAND raster_tests)
add_library(sudoku_generator games/sudoku/src/sudoku_bridge.cpp)
target_link_libraries(sudoku_generator PUBLIC game_rules PRIVATE qjs)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(GAMES_EXTRA_GAME_DIRS "" CACHE STRING "Additional game folders to test before copying into games/")
file(GLOB module_manifests CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/games/*/GAME.json"
  "${CMAKE_CURRENT_SOURCE_DIR}/shared/*/ENGINE.json")
set(catalog_arguments)
foreach(directory IN LISTS GAMES_EXTRA_GAME_DIRS)
  list(APPEND catalog_arguments --extra "${directory}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${directory}/GAME.json")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${module_manifests} "${CMAKE_CURRENT_SOURCE_DIR}/tools/game_catalog.py")
set(GAMES_CATALOG_DIR "${CMAKE_CURRENT_BINARY_DIR}/catalog")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/game_catalog.py"
  --root "${CMAKE_CURRENT_SOURCE_DIR}" --generate "${GAMES_CATALOG_DIR}" ${catalog_arguments}
  RESULT_VARIABLE catalog_result OUTPUT_VARIABLE catalog_output ERROR_VARIABLE catalog_error)
if(NOT catalog_result EQUAL 0)
  message(FATAL_ERROR "Game catalog: ${catalog_error}")
endif()
include_directories("${GAMES_CATALOG_DIR}")
include("${GAMES_CATALOG_DIR}/game_modules.cmake")
add_test(NAME game_catalog COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/game_catalog_tests.py")
if(GAMES_BUILD_APPLICATION)
  include(cmake/Application.cmake)
endif()
add_executable(rules_tests tests/rules_tests.cpp)
target_link_libraries(rules_tests PRIVATE game_rules)
add_test(NAME rules COMMAND rules_tests)
add_library(game_solver shared/cards/solitaire_solver.cpp shared/cards/deal_tables.cpp)
target_link_libraries(game_solver PUBLIC game_rules)
add_executable(solver_tests tests/solver_tests.cpp)
target_link_libraries(solver_tests PRIVATE game_solver)
add_test(NAME solver COMMAND solver_tests)
add_executable(deal_grader EXCLUDE_FROM_ALL tools/deal_grader.cpp)
target_link_libraries(deal_grader PRIVATE game_solver)

add_executable(storage_tests tests/storage_tests.cpp)
target_link_libraries(storage_tests PRIVATE game_rules)
add_test(NAME storage COMMAND storage_tests)
add_executable(settings_tests tests/settings_tests.cpp)
target_link_libraries(settings_tests PRIVATE game_rules)
add_test(NAME settings COMMAND settings_tests)
add_executable(sudoku_tests tests/sudoku_tests.cpp)
target_link_libraries(sudoku_tests PRIVATE sudoku_generator)
target_compile_definitions(sudoku_tests PRIVATE GAMES_TEST_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
add_test(NAME sudoku COMMAND sudoku_tests)

add_executable(puzzle_tests tests/puzzle_tests.cpp)
target_link_libraries(puzzle_tests PRIVATE game_rules)
add_test(NAME puzzles COMMAND puzzle_tests)
foreach(test_target parrots_rules_tests liarsdice_rules_tests penthesheep_rules_tests koikoi_rules_tests rules_tests solver_tests ui_tests storage_tests audio_tests sudoku_tests puzzle_tests collection_ui_tests eggy_sim_tests switchbox_rules_tests switchbox_storage_tests switchbox_actor_tests atomprobe_rules_tests atomprobe_storage_tests fourpegs_rules_tests fourpegs_storage_tests)
  if(NOT TARGET ${test_target})
    continue()
  endif()
  if(MSVC)
    target_compile_options(${test_target} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${test_target} PRIVATE -UNDEBUG)
  endif()
endforeach()

# Keep assertions active in every registered test, including newly discovered modules.
get_property(all_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(test_target IN LISTS all_targets)
  if(test_target MATCHES "_tests$")
    if(MSVC)
      target_compile_options(${test_target} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${test_target} PRIVATE -UNDEBUG)
    endif()
  endif()
endforeach()
