cmake_minimum_required(VERSION 3.25)

  enable_language(OBJCXX)
endif()

  GIT_TAG 8eb27e39848b73885e092777f0a67a4a2d8e5b99)

  vendor/eggy/src/raster.cpp vendor/eggy/src/r3d.cpp vendor/eggy/src/mesh.cpp vendor/eggy/src/textures.cpp
  vendor/eggy/src/ground.cpp vendor/eggy/src/flora.cpp vendor/eggy/src/critters.cpp vendor/eggy/src/duck3d.cpp vendor/eggy/src/scene.cpp)

  vendor/switchbox/src/stage.cpp vendor/switchbox/src/actor.cpp vendor/switchbox/src/save.cpp
  vendor/switchbox/src/mole.cpp)

  vendor/atomprobe/src/platform/mesh.cpp vendor/atomprobe/src/box.cpp
  vendor/atomprobe/src/chamber.cpp vendor/atomprobe/src/save.cpp)
